rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    function isAdmin() {
      return request.auth != null &&
        request.auth.uid == 'PASTE_YOUR_ANONYMOUS_UID_HERE';
    }

    match /products/{productId} {
      allow read: if true;
      allow create, update, delete: if isAdmin();
    }

    match /orders/{orderId} {
      // Checkout is public, but orders are private.
      allow create: if request.resource.data.keys().hasAll([
          'id', 'number', 'date', 'status', 'customer',
          'delivery', 'payment', 'comment', 'items', 'total', 'createdAt'
        ])
        && request.resource.data.keys().hasOnly([
          'id', 'number', 'date', 'status', 'customer',
          'delivery', 'payment', 'comment', 'items', 'total', 'createdAt'
        ])
        && request.resource.data.status == 'Нове'
        && request.resource.data.total is number
        && request.resource.data.total >= 0
        && request.resource.data.items is list
        && request.resource.data.items.size() > 0
        && request.resource.data.items.size() <= 50
        && request.resource.data.customer is map
        && request.resource.data.customer.phone is string
        && request.resource.data.customer.phone.size() <= 30
        && request.resource.data.customer.firstName is string
        && request.resource.data.customer.firstName.size() <= 100
        && request.resource.data.customer.lastName is string
        && request.resource.data.customer.lastName.size() <= 100
        && request.resource.data.createdAt == request.time;
      allow read, update, delete: if isAdmin();
    }
  }
}
