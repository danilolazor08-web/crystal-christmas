rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    // УВАГА: будь-хто з посиланням може змінювати товари й замовлення.
    function canManage() {
      return request.auth != null;
    }
    match /products/{productId} {
      allow read: if true;
      allow create, update, delete: if canManage();
    }
    match /orders/{orderId} {
      allow create: if request.resource.data.keys().hasAll(['number', 'items', 'total'])
        && request.resource.data.keys().hasOnly([
          'id','number','items','total','firstName','lastName','phone',
          'email','city','deliveryMethod','warehouse','payment','comment',
          'status','createdAt'
        ]);
      allow read, update, delete: if canManage();
    }
  }
}
